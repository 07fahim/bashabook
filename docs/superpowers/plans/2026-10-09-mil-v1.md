# Mil V1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Mil V1, the shared-flat app (meals, cook headcount, bazar, bills, balances, month close) described in `docs/PRD.md`.

**Architecture:** One Next.js App Router app in `mil/`, backed by Postgres. All money and meal logic lives in two pure, unit-tested modules (`lib/money.ts`, `lib/meals.ts`). Pages and Server Actions only load rows, call those modules, and render.

**Tech Stack:** Next.js (App Router, TypeScript), Tailwind + shadcn/ui, Postgres via `postgres`, `zod`, `bcryptjs`, `web-push`, Vitest, PGlite (tests only).

## Global Constraints

- Money is integer **poisha** (`number`) everywhere. No floating point in the money path. Meal weights are integer hundredths (`b: 50` = 0.5).
- Dates are `'YYYY-MM-DD'` strings in Asia/Dhaka. Months are identified by their first day `'YYYY-MM-01'`.
- `flatId` always comes from the session, never from the URL or form input (TRD §4).
- `status = 'pending'` rows never affect totals.
- UI copy exists in `bn` and `en`; Bangla is the default.
- Visual tokens and components follow `docs/DESIGN_BRIEF.md`.
- Database = `docs/schema.sql` (copied to `mil/db/migrations/001_init.sql`).

```ts
export type Split = 'equal' | 'meals' | 'room' | 'selected';

/** Split `total` poisha by integer `weights`; shares sum exactly to `total`.
 *  Largest remainder, ties to the lowest index. Null when every weight is 0. */
export function allocate(total: number, weights: number[]): number[] | null {
  const sumW = weights.reduce((a, b) => a + b, 0);
  if (sumW === 0) return null;
  const shares = weights.map(w => Math.floor((total * w) / sumW));
  const order = weights.map((w, i) => ({ i, r: (total * w) % sumW })).sort((a, b) => b.r - a.r || a.i - b.i);
  let left = total - shares.reduce((a, b) => a + b, 0);
  for (const { i } of order) { if (left-- <= 0) break; shares[i]++; }
  return shares;
}
```

```ts
export type Meal = 'b' | 'l' | 'd';
export const MEALS: Meal[] = ['b', 'l', 'd'];

export interface MealMember {
  id: string;
  joinedOn: string;          // 'YYYY-MM-DD'
  leftOn: string | null;     // last day in the flat, inclusive
  defaults: Record<Meal, boolean>;
}

export interface MealChange {
  memberId: string;
  date: string;
  meal: Meal;
  count: number;             // 0 off, 1 on, 2+ with guests
  status: 'ok' | 'pending';
}

/** All dates of the month that starts on `month` ('YYYY-MM-01'). */
export function daysOfMonth(month: string): string[] {
  const [y, m] = month.split('-').map(Number);
  const n = new Date(Date.UTC(y, m, 0)).getUTCDate();
  return Array.from({ length: n }, (_, i) => `${month.slice(0, 8)}${String(i + 1).padStart(2, '0')}`);
}

export function isPresent(m: MealMember, date: string): boolean {
  return date >= m.joinedOn && (m.leftOn === null || date <= m.leftOn);
}

const key = (memberId: string, date: string, meal: Meal) => `${memberId}|${date}|${meal}`;

/** Confirmed changes only; pending ones are ignored so the default applies. */
export function indexChanges(changes: MealChange[]): Map<string, number> {
  const idx = new Map<string, number>();
  for (const c of changes) if (c.status === 'ok') idx.set(key(c.memberId, c.date, c.meal), c.count);
  return idx;
}

export function mealCount(m: MealMember, idx: Map<string, number>, date: string, meal: Meal): number {
  if (!isPresent(m, date)) return 0;
  return idx.get(key(m.id, date, meal)) ?? (m.defaults[meal] ? 1 : 0);
}

/** Weighted meals per member for the month, in hundredths of a meal. Weights are hundredths too (b: 50 = 0.5). */
export function weightedMeals(members: MealMember[], changes: MealChange[], month: string, weights: Record<Meal, number>): Map<string, number> {
  const idx = indexChanges(changes);
  const days = daysOfMonth(month);
  return new Map(members.map(m => [m.id, days.reduce((sum, d) => sum + MEALS.reduce((s, meal) => s + mealCount(m, idx, d, meal) * weights[meal], 0), 0)]));
}

export function daysPresent(members: MealMember[], month: string): Map<string, number> {
  const days = daysOfMonth(month);
  return new Map(members.map(m => [m.id, days.filter(d => isPresent(m, d)).length]));
}

/** Plates per meal for one date, guests included, unweighted. */
export function headcount(members: MealMember[], changes: MealChange[], date: string): Record<Meal, number> {
  const idx = indexChanges(changes);
  const out: Record<Meal, number> = { b: 0, l: 0, d: 0 };
  for (const m of members) for (const meal of MEALS) out[meal] += mealCount(m, idx, date, meal);
  return out;
}

/** A date is locked if it is today or earlier, or it is tomorrow and the Dhaka clock has passed the cutoff.
 *  `nowDhaka` = 'YYYY-MM-DDTHH:MM' in Asia/Dhaka; `cutoff` = 'HH:MM'. */
export function isLocked(date: string, nowDhaka: string, cutoff: string): boolean {
  const [today, time] = nowDhaka.split('T');
  if (date <= today) return true;
  const t = new Date(`${today}T00:00:00Z`);
  t.setUTCDate(t.getUTCDate() + 1);
  return date === t.toISOString().slice(0, 10) && time >= cutoff;
}
```

```ts
export interface LedgerRow {
  id: string;
  type: 'deposit' | 'bazar' | 'bill' | 'payout';
  amount: number;            // poisha, positive integer
  paidBy: string | null;     // bazar/bill: member id, null = paid from fund
  memberId: string | null;   // deposit: payer; payout: receiver
  split: Split | null;       // bills only
  selected: string[];        // split = 'selected'
  status: 'ok' | 'pending';
}

export interface MonthInput {
  memberIds: string[];
  weightedMeals: Map<string, number>;   // hundredths of a meal
  daysPresent: Map<string, number>;
  roomOf: Map<string, string | null>;
  rooms: { id: string; rentShare: number }[];
  ledger: LedgerRow[];
}

export interface MemberResult {
  id: string;
  paidIn: number;
  foodCost: number;
  billShare: number;
  payouts: number;
  balance: number;           // + flat owes them, - they owe
}

export interface MonthResult {
  mealRate: number;          // poisha per weighted meal, rounded, display only
  perMember: MemberResult[];
  fund: number;
  invariantOk: boolean;
  problems: string[];        // reasons the month cannot close
}

export function monthBalances(input: MonthInput): MonthResult {
  const ids = input.memberIds;
  const res = new Map<string, MemberResult>(ids.map(id => [id, { id, paidIn: 0, foodCost: 0, billShare: 0, payouts: 0, balance: 0 }]));
  const problems: string[] = [];
  const rows = input.ledger.filter(l => l.status === 'ok');
  const daysOf = (id: string) => input.daysPresent.get(id) ?? 0;
  const meals = ids.map(id => input.weightedMeals.get(id) ?? 0);
  let fund = 0;

  const charge = (field: 'foodCost' | 'billShare', rowId: string, amount: number, who: string[], weights: number[]) => {
    const shares = allocate(amount, weights);
    if (!shares) { problems.push(`${rowId}: nobody to split this between`); return; }
    shares.forEach((s, i) => { res.get(who[i])![field] += s; });
  };

  const totalBazar = rows.filter(l => l.type === 'bazar').reduce((a, l) => a + l.amount, 0);
  if (totalBazar > 0) charge('foodCost', 'bazar', totalBazar, ids, meals);

  for (const l of rows) {
    if (l.type === 'deposit') { res.get(l.memberId!)!.paidIn += l.amount; fund += l.amount; continue; }
    if (l.type === 'payout') { res.get(l.memberId!)!.payouts += l.amount; fund -= l.amount; continue; }
    if (l.paidBy) res.get(l.paidBy)!.paidIn += l.amount; else fund -= l.amount;
    if (l.type !== 'bill') continue;
    if (l.split === 'equal') charge('billShare', l.id, l.amount, ids, ids.map(daysOf));
    else if (l.split === 'meals') charge('billShare', l.id, l.amount, ids, meals);
    else if (l.split === 'selected') charge('billShare', l.id, l.amount, l.selected, l.selected.map(() => 1));
    else if (l.split === 'room') {
      const inRoom = (roomId: string) => ids.filter(id => input.roomOf.get(id) === roomId && daysOf(id) > 0);
      const occupied = input.rooms.filter(room => inRoom(room.id).length > 0);
      if (occupied.length < input.rooms.length) problems.push(`${l.id}: empty room(s) ${input.rooms.filter(x => !occupied.includes(x)).map(x => x.id).join(', ')}`);
      const roomShares = allocate(l.amount, occupied.map(room => room.rentShare));
      if (!roomShares) { problems.push(`${l.id}: no occupied rooms`); continue; }
      occupied.forEach((room, i) => charge('billShare', l.id, roomShares[i], inRoom(room.id), inRoom(room.id).map(daysOf)));
    }
  }

  for (const m of res.values()) m.balance = m.paidIn - m.foodCost - m.billShare - m.payouts;
  const perMember = [...res.values()];
  const totalMeals = meals.reduce((a, b) => a + b, 0);
  return {
    mealRate: totalMeals ? Math.round((totalBazar * 100) / totalMeals) : 0,
    perMember,
    fund,
    invariantOk: problems.length === 0 && perMember.reduce((a, m) => a + m.balance, 0) === fund,
    problems,
  };
}
```

```ts
import { describe, it, expect } from 'vitest';
import { PGlite } from '@electric-sql/pglite';
import { readFileSync, readdirSync } from 'node:fs';
import { join } from 'node:path';

const dir = join(__dirname, 'migrations');

async function freshDb() {
  const db = new PGlite();
  for (const f of readdirSync(dir).filter(f => f.endsWith('.sql')).sort()) await db.exec(readFileSync(join(dir, f), 'utf8'));
  const flat = (await db.query<{ id: string }>(`insert into flats (name, join_code) values ('x', 'abc') returning id`)).rows[0].id;
  const mgr = (await db.query<{ id: string }>(`insert into members (flat_id, name, joined_on, role) values ($1, 'a', '2026-10-01', 'manager') returning id`, [flat])).rows[0].id;
  await db.query(`insert into months (flat_id, month) values ($1, '2026-10-01')`, [flat]);
  return { db, flat, mgr };
}

describe('schema', () => {
  it('allows only one manager per flat', async () => {
```

- [ ] **Step 4: Write the migration runner** (`mil/db/migrate.ts`)

```ts
import postgres from 'postgres';
import { readFileSync, readdirSync } from 'node:fs';
import { join } from 'node:path';

const sql = postgres(process.env.DATABASE_URL!, { max: 1 });
const dir = join(__dirname, 'migrations');

async function main() {
  await sql`create table if not exists migrations (name text primary key, applied_at timestamptz not null default now())`;
  const done = new Set((await sql<{ name: string }[]>`select name from migrations`).map(r => r.name));
  for (const name of readdirSync(dir).filter(f => f.endsWith('.sql')).sort()) {
    if (done.has(name)) continue;
    await sql.begin(async tx => {
      await tx.unsafe(readFileSync(join(dir, name), 'utf8'));
      await tx`insert into migrations (name) values (${name})`;
    });
    console.log('applied', name);
  }
}

main().finally(() => sql.end());
```
